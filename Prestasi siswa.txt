    const belum = list.filter(m => m.dari !== 'siswa' && !m.dibaca);
    const c = $('#pr-unread'); c.hidden = !belum.length; c.textContent = `${belum.length} baru`;
    return belum;
  }
  // tandai pesan guru sebagai dibaca ketika kotak umpan balik terlihat di layar
  let terlihat = false;
  new IntersectionObserver(es => { terlihat = es.some(e => e.isIntersecting); if (terlihat) tandaiDibaca(); }, { threshold: .3 }).observe(document.getElementById('pr-thread'));
  function tandaiDibaca() {
    if (!API || !UID || !terlihat || document.visibilityState !== 'visible') return;
    const up = {};
    Object.entries(S.umpan).forEach(([id, m]) => { if (m.dari !== 'siswa' && !m.dibaca) up[`${id}/dibaca`] = true; });
    if (Object.keys(up).length) API.D.update(API.D.ref(API.db, `umpanBalik/${UID}`), up).catch(() => { });
  }

  async function kirim() {
    const ta = $('#pr-text'), teks = ta.value.trim();
    if (!teks || !API || !UID) return;
    const btn = $('#pr-send'); btn.disabled = true;
    try {
      const { D, db } = API;
      await D.set(D.push(D.ref(db, `umpanBalik/${UID}`)), { teks, waktu: D.serverTimestamp(), konteks: 'umum', dari: 'siswa', dibaca: false });
      D.set(D.push(D.ref(db, 'aktivitas')), { uid: UID, jenis: 'umpan', pesan: 'mengirim pesan ke guru', waktu: D.serverTimestamp() }).catch(() => { });
      ta.value = '';
    } catch (e) { toast('Pesan gagal terkirim. Periksa koneksi internet.'); }
    btn.disabled = !API;
  }

  function renderAll() { renderStats(); renderChart(); renderBadges(); }

  /* ---------- event ---------- */
  document.querySelectorAll('#prestasi [data-g]').forEach(b => b.addEventListener('click', () => {
    S.grafik = b.dataset.g;
    document.querySelectorAll('#prestasi [data-g]').forEach(x => x.setAttribute('aria-pressed', String(x === b)));
    renderChart();
  }));
  $('#pr-send').addEventListener('click', kirim);
  $('#pr-text').addEventListener('keydown', e => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); kirim(); } });
  document.addEventListener('visibilitychange', tandaiDibaca);
  addEventListener('ek-belajar', e => { const k = key(new Date()); S.harian[k] = (+S.harian[k] || 0) + e.detail.detik; renderAll(); });
  window.__prestasiDemo = (data) => { Object.assign(S, data); renderAll(); renderUmpan(); }; // untuk uji tampilan

  renderAll();

  /* ---------- koneksi Firebase (real-time) ---------- */
  try {
    const V = '10.12.2';
    const APP = await import(`https://www.gstatic.com/firebasejs/${V}/firebase-app.js`);
    const A = await import(`https://www.gstatic.com/firebasejs/${V}/firebase-auth.js`);
    const D = await import(`https://www.gstatic.com/firebasejs/${V}/firebase-database.js`);
    const app = APP.getApps().length ? APP.getApp() : APP.initializeApp(FIREBASE_CONFIG);
    const db = D.getDatabase(app);
    D.onValue(D.ref(db, '.info/connected'), s => live(!!s.val() && !!UID, s.val() ? (UID ? 'Terhubung langsung' : 'Belum masuk') : 'Menyambung ulang…'));
    A.onAuthStateChanged(A.getAuth(app), (user) => {
      if (!user) {
        live(false, 'Belum masuk');
        $('#pr-thread').innerHTML = '<div class="pr-empty"><b>🔒</b>Masuk dulu untuk melihat prestasi dan pesan dari guru.<br><a href="loging.html" style="color:#0f2a44;font-weight:800">Masuk sekarang</a></div>';
        return;
      }
      UID = user.uid; API = { D, db };
      $('#pr-send').disabled = false;
      live(true, 'Terhubung langsung');
      const dengar = (path, fn) => D.onValue(D.ref(db, path), s => { fn(s.val() || {}); }, err => console.warn(path, err));
      dengar(`belajarHarian/${UID}`, v => { S.harian = v; renderAll(); });
      dengar(`progres_siswa/${UID}`, v => { S.progres = v; renderAll(); });
      dengar(`kuis/${UID}`, v => { S.kuis = v; renderAll(); });
      dengar(`lab/${UID}`, v => { S.lab = v; renderAll(); });
      dengar(`umpanBalik/${UID}`, v => {
        const lama = new Set(Object.keys(S.umpan));
        S.umpan = v;
        const baru = Object.entries(v).filter(([id, m]) => !lama.has(id) && m.dari !== 'siswa');
        if (!pertamaUmpan && baru.length) toast('💬 Pesan baru dari guru: ' + String(baru[baru.length - 1][1].teks).slice(0, 80));
        renderUmpan(); renderBadges(); tandaiDibaca();
        pertamaUmpan = false;
      });
    });
  } catch (e) {
    live(false, 'Tidak terhubung');
    $('#pr-thread').innerHTML = '<div class="pr-empty"><b>📡</b>Tidak dapat terhubung ke server.</div>';
  }
</script>
